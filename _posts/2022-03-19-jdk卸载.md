# 一 检查安装的jdk
java -version

```git
[root@localhost bin]# java -version
java version "1.8.0_121"
Java(TM) SE Runtime Environment (build 1.8.0_121-b13)
Java HotSpot(TM) 64-Bit Server VM (build 25.121-b13, mixed mode)
```

# 二 检查jdk安装目录
which java

```git
[root@localhost bin]# which java
/usr/local/jdk1.8.0_121/bin/java
```

# 三 删除安装文件
rm -ef [文件目录]

```git
rm -rf /usr/local/jdk1.8.0_121/
```

# 四 删除环境变量配置
vim /etc/profile

```git
[root@localhost local]# vim /etc/profile

export JAVA_HOME=/usr/local/jdk1.8.0_121 
export PATH=$JAVA_HOME/bin:$PATH 
export CLASSPATH=.:$JAVA_HOME/lib/dt.jar:$JAVA_HOME/lib/tools.jar:$JAVA_HOME/jre/lib/rt.jar 
```

# 六 卸载自带的jdk
查询：

<font style="background-color:#FFFFFF;">rpm -qa |grep java</font>

<font style="background-color:#FFFFFF;">rpm -qa |grep jdk</font>

<font style="background-color:#FFFFFF;">rpm -qa |grep gcj</font>

<font style="background-color:#FFFFFF;">卸载，</font><font style="color:rgb(57, 57, 57);background-color:#FFFFFF;">批量卸载所有带有Java的文件  关键字是java</font>

<font style="background-color:#FFFFFF;">rpm -qa | grep java | xargs rpm -e --nodeps</font><font style="color:rgb(57, 57, 57);background-color:#FFFFFF;"> </font>

```git

[root@localhost local]# rpm -qa|grep java
tzdata-java-2016g-2.el7.noarch
python-javapackages-3.4.1-11.el7.noarch
[root@localhost local]# rpm -qa | grep java | xargs rpm -e --nodeps
[root@localhost local]# rpm -qa|grep java
[root@localhost local]# rpm -qa|grep java
[root@localhost local]# rpm -qa|grep jdk
[root@localhost local]# rpm -qa|grep gcj

```
