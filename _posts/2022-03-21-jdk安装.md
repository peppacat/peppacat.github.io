# 一 下载jdk 
[下载：jdk-11.0.13_linux-x64_bin.tar.gz](https://www.123pan.com/s/94VA-ZFEVv)

# 二  用WinSCP上传到centos7服务器上
# 三 新建目录
mkdir 【目录】

```git
mkidr /usr/local/java
```

# 四 解压文件
tar -zxvf -C 【指定目录】

```git
tar -zxvf jdk-11.0.13_linux-x64_bin.tar.gz -C /usr/local/java/
```

# 五 添加环境变量
vim /etc/profile 文件最后加入环境变量配置（jdk11集成了jre，不需要配置jre）

1. cp /etc/profile  /etc/profile-bnk 备份配置文件
2. vim /etc/profile
3. 【i】进入编辑模式
4. 【Shift＋Ins】复制环境变量配置
5. 【Esc】退出编辑模式
6. 【：wq】保存/ 【:q!】退出不保存

```java
# java 环境变量配置
export JAVA_HOME=/usr/local/java/jdk-11.0.13
export PATH=$JAVA_HOME/bin:$PATH
```

```java
export JAVA_HOME=/usr/local/jdk1.8.0_121 
export PATH=$JAVA_HOME/bin:$PATH 
export CLASSPATH=.:$JAVA_HOME/lib/dt.jar:$JAVA_HOME/lib/tools.jar:$JAVA_HOME/jre/lib/rt.jar
```

```git
vim /etc/profile



if [ $UID -gt 199 ] && [ "`/usr/bin/id -gn`" = "`/usr/bin/id -un`" ]; then
    umask 002
else
    umask 022
fi

for i in /etc/profile.d/*.sh ; do
    if [ -r "$i" ]; then
        if [ "${-#*i}" != "$-" ]; then
            . "$i"
        else
            . "$i" >/dev/null
        fi
    fi
done

unset i
unset -f pathmunge

# java 环境变量配置
export JAVA_HOME=/usr/local/java/jdk-11.0.13

```

# 六 配置立即生效
source /etc/profile 

```java
source /etc/profile
```
